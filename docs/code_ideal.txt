```App.tsx
import React, { useState } from 'react'
import { Onboarding } from './screens/Onboarding'
import { Home } from './screens/Home'
import { ProductDetail } from './screens/ProductDetail'
import { SellerDashboard } from './screens/SellerDashboard'
import { ProductManagement } from './screens/ProductManagement'
import { InterestManagement } from './screens/InterestManagement'
import { NavigationBar } from './components/NavigationBar'
import { Header } from './components/Header'
export function App() {
  const [currentScreen, setCurrentScreen] = useState('onboarding')
  const renderScreen = () => {
    switch (currentScreen) {
      case 'onboarding':
        return <Onboarding onComplete={() => setCurrentScreen('home')} />
      case 'home':
        return (
          <Home onProductSelect={() => setCurrentScreen('productDetail')} />
        )
      case 'productDetail':
        return <ProductDetail onBack={() => setCurrentScreen('home')} />
      case 'sellerDashboard':
        return (
          <SellerDashboard
            onProductManage={() => setCurrentScreen('productManagement')}
          />
        )
      case 'productManagement':
        return (
          <ProductManagement
            onBack={() => setCurrentScreen('sellerDashboard')}
          />
        )
      case 'interestManagement':
        return (
          <InterestManagement
            onBack={() => setCurrentScreen('sellerDashboard')}
          />
        )
      default:
        return <Home />
    }
  }
  const showNavigation = currentScreen !== 'onboarding'
  const showHeader = currentScreen !== 'onboarding'
  return (
    <div className="flex flex-col h-screen w-full bg-gray-50">
      {showHeader && (
        <Header
          currentScreen={currentScreen}
          setCurrentScreen={setCurrentScreen}
        />
      )}
      <div className="flex-1 overflow-y-auto">{renderScreen()}</div>
      {showNavigation && (
        <NavigationBar
          currentScreen={currentScreen}
          setCurrentScreen={setCurrentScreen}
        />
      )}
    </div>
  )
}
```

```components/Header.tsx
import React from 'react'
import { ArrowLeftIcon, BellIcon, SearchIcon } from 'lucide-react'
interface HeaderProps {
  currentScreen: string
  setCurrentScreen: (screen: string) => void
}
export function Header({ currentScreen, setCurrentScreen }: HeaderProps) {
  const renderTitle = () => {
    switch (currentScreen) {
      case 'home':
        return 'Marketplace'
      case 'productDetail':
        return 'Product Details'
      case 'sellerDashboard':
        return 'Seller Dashboard'
      case 'productManagement':
        return 'Manage Products'
      case 'interestManagement':
        return 'Manage Interests'
      default:
        return 'Marketplace'
    }
  }
  const showBackButton = [
    'productDetail',
    'productManagement',
    'interestManagement',
  ].includes(currentScreen)
  const handleBack = () => {
    if (currentScreen === 'productDetail') {
      setCurrentScreen('home')
    } else if (
      currentScreen === 'productManagement' ||
      currentScreen === 'interestManagement'
    ) {
      setCurrentScreen('sellerDashboard')
    }
  }
  return (
    <div className="bg-white border-b border-gray-200 px-4 py-3 flex items-center justify-between">
      <div className="flex items-center">
        {showBackButton && (
          <button onClick={handleBack} className="mr-3">
            <ArrowLeftIcon className="w-6 h-6 text-gray-700" />
          </button>
        )}
        <h1 className="font-bold text-lg text-gray-800">{renderTitle()}</h1>
      </div>
      <div className="flex items-center space-x-4">
        {currentScreen === 'home' && (
          <button className="p-1">
            <SearchIcon className="w-6 h-6 text-gray-700" />
          </button>
        )}
        <button className="p-1 relative">
          <BellIcon className="w-6 h-6 text-gray-700" />
          <span className="absolute top-0 right-0 w-2 h-2 bg-red-500 rounded-full"></span>
        </button>
      </div>
    </div>
  )
}
```

```components/Button.tsx
import React from 'react'
interface ButtonProps {
  label: string
  onClick?: () => void
  variant?: 'primary' | 'secondary' | 'outline'
  fullWidth?: boolean
  icon?: React.ReactNode
}
export function Button({
  label,
  onClick,
  variant = 'primary',
  fullWidth = false,
  icon,
}: ButtonProps) {
  const baseStyles =
    'flex items-center justify-center px-4 py-3 rounded-lg font-medium'
  const variantStyles = {
    primary: 'bg-blue-500 text-white',
    secondary: 'bg-blue-100 text-blue-700',
    outline: 'bg-white border border-gray-300 text-gray-700',
  }
  const widthStyles = fullWidth ? 'w-full' : ''
  return (
    <button
      className={`${baseStyles} ${variantStyles[variant]} ${widthStyles}`}
      onClick={onClick}
    >
      {icon && <span className="mr-2">{icon}</span>}
      {label}
    </button>
  )
}
```
